// Règles de sécurité Firestore recommandées pour Rami 51
// À copier dans la console Firebase > Firestore Database > Règles

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Rooms : lecture publique, écriture authentifiée
    match /rooms/{roomId} {
      allow read: if true;
      allow write: if request.auth != null;
      
      // Events dans une room
      match /events/{eventId} {
        allow read: if true;
        allow create: if request.auth != null;
        allow update, delete: if false; // Events sont immutables
      }
    }
    
    // Par défaut : tout bloquer
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

// Instructions pour activer:
// 1. Allez sur https://console.firebase.google.com/
// 2. Sélectionnez votre projet
// 3. Firestore Database > Règles
// 4. Copiez-collez ces règles
// 5. Cliquez sur "Publier"
