// Règles de sécurité Firestore minimum viable pour Rami 51
// À copier dans la console Firebase > Firestore Database > Règles

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Rooms : lecture/écriture authentifiée uniquement
    match /rooms/{roomId} {
      allow read, write: if request.auth != null;
      
      // Events dans une room : authentifié uniquement
      match /events/{eventId} {
        allow read, write: if request.auth != null;
      }
    }
  }
}

// Instructions pour activer:
// 1. Allez sur https://console.firebase.google.com/
// 2. Sélectionnez votre projet
// 3. Firestore Database > Règles
// 4. Copiez-collez ces règles
// 5. Cliquez sur "Publier"
//
// NOTE: Ces règles nécessitent que les utilisateurs soient authentifiés.
// Pour un MVP sans auth, vous pouvez temporairement utiliser:
//   allow read, write: if true;
// Mais attention, cela expose vos données publiquement!
